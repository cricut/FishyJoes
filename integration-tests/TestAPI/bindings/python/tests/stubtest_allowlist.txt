# Intentional dynamic runtime details that are not part of the public typing
# surface exposed to IDEs/type checkers.

# The runtime package's public API is re-exported through fishyjoes_runtime.
# The implementation module defines descriptor classes inside create_runtime(),
# so stubtest sees implementation-only names that are not import-level runtime
# objects. Keep these out of the public package stubs.
fishyjoes_runtime.__all__
fishyjoes_runtime.RuntimeConfig.__match_args__
fishyjoes_runtime.RuntimeConfig.__replace__
fishyjoes_runtime.RuntimeDependency.__match_args__
fishyjoes_runtime.RuntimeDependency.__replace__
fishyjoes_runtime.iota..*
# Same pattern: the Foundation AttributedString family classes are defined inside
# setup_attributed_string_family(), so stubtest sees implementation-only locals
# that are not import-level runtime objects. The public surface is re-exported as
# the typed testapi._native.Runtime_* classes (verified against the runtime
# elsewhere in this file).
fishyjoes_runtime.attributed_string..*

# __all__ is computed from generated exports at runtime.
testapi.__all__
testapi._exports.__all__

# Swift-origin metadata is runtime diagnostics data, not part of the typed
# public API.
testapi\..*\.__fishyjoes_origin__

# Python cannot declare `AssociatedDataEnum.None` in a stub because `None` is a
# keyword, even though the runtime exposes it via setattr for Swift case parity.
testapi.associated_data_enum.AssociatedDataEnum.None

# Static-property descriptors use runtime metaclasses; stubs model the useful
# user-facing attribute types instead of the descriptor implementation class.
testapi.collections__collection_holder.Collections_CollectionHolder
testapi.deprecations.Deprecations
testapi.deprecations.Deprecations.deprecated_variable
testapi.methods.Methods
testapi.primitives__primitive_holder.Primitives_PrimitiveHolder

# Range/Result value-type members are the same StaticProperty-descriptor case:
# the stub models the precise user-facing attribute type (SwiftRange[int],
# SwiftClosedRange[float], ResultSuccess[..]|ResultFailure[..]), but the
# descriptor returns a frozen-dataclass instance whose generic parameter is
# unreified at runtime (SwiftRange[Any]), which stubtest cannot match against
# the parameterised stub. The precise stub is the contract and is independently
# gated by test_range_and_result_types_are_precise and pyright --verifytypes.
testapi.ranges.Ranges.(int|int8|int16|int32|int64|u_int|u_int8|u_int16|u_int32|u_int64)_range
testapi.closed_ranges.ClosedRanges.(int|int8|int16|int32|int64|u_int|u_int8|u_int16|u_int32|u_int64|double|float|string)_range
testapi.results.Results.(a_success|a_failure)

# CPython reports object.__eq__ as positional-only. The stub keeps the normal
# `other: object` shape expected by IDEs and type checkers.
testapi.reference_only_types__marker.ReferenceOnlyTypes_Marker.__eq__
