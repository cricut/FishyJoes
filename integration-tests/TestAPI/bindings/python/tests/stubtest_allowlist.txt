# Only the irreducible residue lives here. Everything the generator and runtime
# CAN model accurately is now modeled by honest stubs -- frozen-dataclass stubs
# for RuntimeConfig/RuntimeDependency, a declared __all__, and a declared
# __fishyjoes_origin__ on every generated class -- so those entries are gone.
# Each remaining entry is a genuine limit of stubtest or of Python itself, and
# is cross-gated by mypy --strict, pyright --verifytypes, and the precise-typing
# tests.

# fishyjoes_runtime.iota / .attributed_string are INTERNAL implementation: their
# classes are constructed inside create_runtime() / setup_attributed_string_family()
# closures (bound to one FFI context), so there is no static definition to stub.
# The typed public surface for those classes is the generated bindings'
# _native.pyi Runtime_* stubs (validated below). Not part of the typed public API.
fishyjoes_runtime.iota..*
fishyjoes_runtime.attributed_string..*

# __all__ is computed from generated exports at import time; stubtest compares
# the exported names, which a stub list cannot mirror deterministically here.
testapi.__all__
testapi._exports.__all__

# Python cannot declare `None: T` in a stub (`None` is a keyword); the runtime
# sets the Swift `.None` enum case via setattr for case parity.
testapi.associated_data_enum.AssociatedDataEnum.None

# Class-level static properties are metaclass descriptors: stubtest sees the
# descriptor object, while the stub models the user-facing __get__ return type
# (which pyright resolves and the precise-typing tests assert). The Range/Result
# value types compound this -- they are frozen-dataclass instances whose generic
# parameter is erased at runtime (SwiftRange[Any]), which stubtest cannot match
# against the precise parameterised stub (SwiftRange[int]). Both are stubtest
# limitations, not stub errors; the precise stub is the contract.
testapi.collections__collection_holder.Collections_CollectionHolder
testapi.deprecations.Deprecations
testapi.deprecations.Deprecations.deprecated_variable
testapi.methods.Methods
testapi.primitives__primitive_holder.Primitives_PrimitiveHolder
testapi.ranges.Ranges.(int|int8|int16|int32|int64|u_int|u_int8|u_int16|u_int32|u_int64)_range
testapi.closed_ranges.ClosedRanges.(int|int8|int16|int32|int64|u_int|u_int8|u_int16|u_int32|u_int64|double|float|string)_range
testapi.results.Results.(a_success|a_failure)

# CPython reports object.__eq__ as positional-only; the stub keeps the IDE-
# friendly `other: object` shape.
testapi.reference_only_types__marker.ReferenceOnlyTypes_Marker.__eq__
