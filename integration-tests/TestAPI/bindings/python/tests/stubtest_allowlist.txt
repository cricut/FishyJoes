# Only the irreducible residue lives here. Everything the generator and runtime
# CAN model accurately is modeled by honest stubs -- frozen-dataclass stubs for
# RuntimeConfig/RuntimeDependency, a declared __all__, and a declared
# __fishyjoes_origin__ on every generated class. Each remaining entry is a
# genuine limit of stubtest or of Python itself, and this run is strict: an
# unused entry fails the gate and must be re-justified or removed.
#
# The fishyjoes_runtime package is stubtested separately with its own allowlist
# (python-runtime/stubtest_allowlist.txt, run by test_runtime_typing.py).

# Python cannot declare `None: T` in a stub (`None` is a keyword); the runtime
# sets the Swift `.None` enum case via setattr for case parity.
testapi.associated_data_enum.AssociatedDataEnum.None

# Class-level static properties are metaclass descriptors: stubtest sees the
# descriptor object, while the stub models the user-facing __get__ return type
# (which pyright resolves and the precise-typing tests assert against the stub
# text). A descriptor is how Python spells a class-level computed property, so
# these cannot go away.
testapi.collections__collection_holder.Collections_CollectionHolder
testapi.deprecations.Deprecations
testapi.deprecations.Deprecations.deprecated_variable
testapi.methods.Methods
testapi.primitives__primitive_holder.Primitives_PrimitiveHolder
